\section{1. 一句话本质}

\begin{bluebox}
\textbf{电、磁、热性能的根,全在电子结构(Ch1 的电子排布)}。\textbf{能带理论把它定量化,
DFT 把它从第一性原理算出来}。

\vspace{0.3em}

\textbf{三句话记住这一章}:

\textbf{1. 电学看带隙}:带隙 = 0 是导体,0-3 eV 是半导体,> 3 eV 是绝缘体。

\textbf{2. 磁学看未成对电子}:3d/4f 的未成对电子产生磁矩(Ch1 的 Fe 3d⁶!),铁磁有居里温度。

\textbf{3. 热学看声子 + 电子}:金属靠电子导热(导电的也导热),绝缘体靠声子(晶格振动)。

\vspace{0.3em}

\textbf{实测核心 Aha}:\textbf{Si 带隙 1.12 eV(半导体),金刚石 5.47 eV(绝缘体)——
带隙这一个数决定了导电性};\textbf{Fe 的 3d⁶ 未成对电子(Ch1 算过!)产生 2.2 $\mu_B$ 磁矩,
居里温度 1043K};\textbf{金刚石不导电,热导却是金属的几倍——靠声子}。

\section{3. 但其实是什么意思}

\subsection{能带:原子能级的"展宽"(从 Ch1 长出来)}

\textbf{Ch1 讲单个原子有分立的能级(1s、2s、2p...)}。\textbf{当 $10^{23}$ 个原子聚成固体,
泡利不相容原理不允许电子占据相同状态——分立能级"展宽"成连续的"能带"}。

\begin{itemize}
\item \textbf{价带}:被电子填满的带(来自成键电子)
\item \textbf{导带}:空的或半满的带(电子能自由移动)
\item \textbf{带隙}:价带顶到导带底的能量间隙
\end{itemize}

\textbf{导电的本质}:\textbf{电子要导电,必须能"加速"(获得能量进入更高能态)}。
\begin{itemize}
\item \textbf{金属}:导带半满,电子随便动 → \textbf{导电}
\item \textbf{半导体}:带隙小(<3 eV),热激发能让少量电子跨过 → \textbf{弱导电}
\item \textbf{绝缘体}:带隙大,电子跨不过去 → \textbf{不导电}
\end{itemize}

\textbf{这就是 Ch1 那个"为什么金导电、玻璃绝缘"的定量答案}——\textbf{差别在带隙}。

\subsection{磁性:回到 Fe 的 3d⁶(首尾呼应)}

\textbf{Ch1 我们算过 Fe 的电子排布是 ...3d⁶4s²}。\textbf{当时只说"3d 电子是磁性来源",
现在我们讲清楚}:

\textbf{3d 轨道有 5 个,按洪特规则(Ch1),6 个 3d 电子的排布是
↑↓ ↑ ↑ ↑ ↑——有 4 个未成对电子}。\textbf{每个未成对电子贡献一个磁矩(自旋)}。
\textbf{在铁磁materials里,这些原子磁矩通过交换作用自发平行排列 → 宏观磁性}。

\textbf{居里温度 $T_c$}:\textbf{温度升高,热运动破坏磁矩的有序排列}。\textbf{超过 $T_c$,
磁矩变无序,铁磁性消失变成顺磁}。\textbf{铁的 $T_c=1043$K——所以铁加热到红热以上就没磁性了}。

\begin{bluebox}
\textbf{首尾呼应的时刻}:\textbf{Ch1 我们用代码算出 Fe = 3d⁶,说"这是磁性来源";
14 章之后,我们终于讲清楚——3d⁶ 的 4 个未成对电子,产生 2.2 $\mu_B$ 磁矩,
通过交换作用排列成铁磁,居里温度 1043K}。\textbf{全书绕了一大圈,回到了电子排布}。
\end{bluebox}

\subsection{热学:声子与电子的分工}

\textbf{热怎么传导}?——\textbf{两种载体}:
\begin{itemize}
\item \textbf{电子}:金属里的自由电子既导电又导热(\textbf{Wiedemann-Franz 定律}:$\kappa = L\sigma T$)
\item \textbf{声子}:晶格振动的"量子"——\textbf{绝缘体靠它导热}
\end{itemize}

\textbf{反直觉的事实}:\textbf{金刚石不导电(绝缘体,带隙 5.47 eV),但热导率是铜的好几倍}!
\textbf{因为金刚石的共价键极强、晶格极规整,声子传播畅通无阻}。\textbf{这就是为什么
金刚石/氮化硼用作高功率电子器件的散热衬底}——\textbf{绝缘但导热}。

\textbf{热容}:\textbf{高温下每个原子的振动贡献固定热容,摩尔热容 $\approx 3R$(杜隆-珀蒂)}。
\textbf{低温下声子"冻结",热容 $\propto T^3$(德拜)}。

\subsection{DFT/MD/MLIP:把性质算出来}

\textbf{今天,这些性质不用只靠实验测,可以从第一性原理算}:
\begin{itemize}
\item \textbf{DFT(密度泛函理论)}:解电子的薛定谔方程(Ch1)的工程化版本——\textbf{算能带、带隙、磁矩}
\item \textbf{MD(分子动力学)}:模拟原子运动——\textbf{算热输运、扩散、热膨胀}
\item \textbf{MLIP(机器学习势)}:用神经网络拟合 DFT,\textbf{兼具 DFT 精度和 MD 速度}
\end{itemize}

\section{4. 真正的数学}

\subsection{本征载流子浓度}

$$n_i \propto \exp\left(-\frac{E_g}{2k_BT}\right)$$

\textbf{带隙越小、温度越高,载流子越多}。\textbf{这解释了半导体器件的温度敏感性}。

\subsection{居里-外斯定律}

$$\chi = \frac{C}{T - T_c} \quad (T > T_c)$$

\textbf{$T > T_c$ 顺磁(磁化率随温度下降);$T < T_c$ 铁磁(自发磁化)}。

\subsection{Wiedemann-Franz 定律}

$$\frac{\kappa_e}{\sigma} = L\,T, \quad L = 2.44\times10^{-8}\ \text{W}\cdot\Omega/\text{K}^2$$

\textbf{金属中电子热导与电导成正比}——\textbf{导电好的金属通常也导热好(铜、银)}。

\subsection{DFT 的核心方程}

\textbf{Kohn-Sham 方程}(把多电子问题化为单电子有效势):
$$\left[-\frac{\hbar^2}{2m}\nabla^2 + V_{\text{eff}}(\mathbf{r})\right]\psi_i = \varepsilon_i\psi_i$$

\textbf{这是 Ch1 薛定谔方程的可计算版本}——\textbf{能量泛函对电子密度求极小,算出基态}。
\textbf{从这里能得到能带、磁矩、形成能(Ch5 的 $H$)、弹性常数(Ch13 的 $E$)}。

\section{5. 一个让人 "Aha" 的例子}

\subsection{核心 Aha:带隙这一个数,决定导电性}

运行配套模块 \texttt{electronic\_properties.py}:

\begin{verbatim}
材料          带隙Eg(eV)   类型
铜 Cu         0.00        导体(金属)

硅 Si         1.12        半导体

\end{verbatim}

\textbf{Aha}:\textbf{一个数——带隙——就决定了材料是导体、半导体还是绝缘体}。\textbf{这正是
Ch1 那句"为什么金导电、玻璃绝缘"的定量答案}。\textbf{硅的带隙 1.12 eV 恰到好处
(室温有少量载流子,可调控)——这就是整个半导体工业建立在硅上的原因}。
\textbf{金刚石带隙 5.47 eV,电子跨不过去,所以绝缘}。

\subsection{Aha:Fe 的磁性,来自 Ch1 的 3d⁶}

\begin{verbatim}
材料    居里温度Tc(K)   磁矩(μB)   电子排布
Fe 铁   1043           2.2       3d⁶4s²
Co 钴   1388           1.7       3d⁷4s²
Ni 镍    627           0.6       3d⁸4s²
\end{verbatim}

\textbf{Aha}:\textbf{Ch1 我们用代码算出 Fe = 3d⁶,说这是磁性来源}。\textbf{现在闭环}:
\textbf{3d⁶ 的未成对电子产生 2.2 $\mu_B$ 磁矩,通过交换作用排成铁磁,居里温度 1043K}。
\textbf{加热到 1043K 以上,热运动打乱磁矩,铁就没磁性了}。\textbf{从 Ch1 的"电子排布"
到 Ch14 的"宏观磁性"——全书在这里首尾相接}。

\subsection{Aha:金刚石不导电,却导热最好}

\begin{verbatim}
材料        电子热导(W/mK)   机制
铜(金属)     436            电子主导
铝(金属)     276            电子主导
金刚石(绝缘)   0(电子)        声子主导(却最高!)
\end{verbatim}

\textbf{Aha}:\textbf{金刚石是绝缘体(带隙 5.47 eV,电子不导电),但它的热导率比铜还高几倍}!
\textbf{秘密在声子}:\textbf{金刚石的共价键极强、晶格极完美,晶格振动(声子)畅通无阻}。
\textbf{所以高功率芯片用金刚石/氮化硼散热——既绝缘又导热}。\textbf{这彻底打破"导电=导热"的直觉}——
\textbf{电子和声子是两种不同的热载体}。

\subsection{怎么看见它:能谱、磁测量与 DFT 验证}

\begin{bluebox}
\textbf{怎么看见它}。\textbf{电子结构靠"谱学"测,磁性靠"磁强计"测——
而 DFT 计算结果要和这些实验对照验证}。
\end{bluebox}

\textbf{探测电子结构和性能的手段}:
\begin{itemize}
\item \textbf{角分辨光电子能谱(ARPES)}:直接"拍"出能带结构——\textbf{把电子的能量-动量关系画出来}
\item \textbf{电输运测量}:四探针测电阻率、霍尔效应测载流子浓度和类型(n/p)
\item \textbf{磁强计(VSM/SQUID)}:测磁矩、磁化曲线、居里温度——\textbf{验证 DFT 算的磁矩}
\item \textbf{X 射线吸收谱(XAS/XMCD)}:测元素分辨的电子态和磁矩
\item \textbf{热分析}:激光闪射法测热导、DSC(Ch5)测热容
\end{itemize}

\textbf{DFT 的验证}:\textbf{DFT 算出的能带、带隙、磁矩,与 ARPES、磁强计的实验对照}——
\textbf{一致才说明计算可信}。\textbf{能算什么}:能带/带隙、磁矩/居里温度、形成能(给 CALPHAD Ch7)、
弹性常数(给 FEM Ch13)、声子谱(给热容/热导)、缺陷能(给 Ch3)。\textbf{局限}:
\textbf{标准 DFT 低估带隙(需 GW/杂化泛函修正)};\textbf{强关联体系(某些氧化物)困难};
\textbf{计算量大(~百原子),大体系靠 MLIP}。

\textbf{现代视角(你的领域)}:\textbf{机器学习势(MLIP)用神经网络拟合 DFT 数据,
能以接近 DFT 的精度模拟百万原子、纳秒尺度——让第一性原理"放大"到介观}。
\textbf{universal MLIP(通用机器学习势)更是能跨元素体系预测——这是当代计算材料学最热的前沿}。

\section{6. 这玩意儿现在在哪}

\subsection{在材料工程中}

\begin{itemize}

\item \textbf{磁性材料}:永磁体(钕铁硼)、软磁(变压器硅钢)、磁存储

\end{itemize}

\subsection{在计算材料学中(本书的现代视角)}

\begin{itemize}
\item \textbf{DFT}:算电子结构、磁矩、形成能——\textbf{材料计算的基石(多尺度的原子一环)}
\item \textbf{MD}:算热输运、扩散、力学(配合 MLIP)
\item \textbf{MLIP}:机器学习势,DFT 精度 + MD 速度——\textbf{当代前沿}
\item \textbf{高通量 + AI}:扫描材料空间,发现新功能材料(Materials Project 等)
\end{itemize}

\begin{bluebox}
\textbf{多尺度计算暗线的"原子尺度"一环在此}:\textbf{DFT/MD/MLIP(原子,本章)→
CALPHAD/相场(微观,Ch7/Ch12)→ FEM(宏观,Ch13)}。\textbf{三个尺度贯穿全书,
现在全部到齐}。\textbf{DFT 算的形成能喂给 CALPHAD(Ch7),算的弹性常数喂给 FEM(Ch13),
算的界面能喂给相场(Ch12)}——\textbf{原子尺度是整个多尺度链条的源头}。
\textbf{而 MLIP 正让这个源头"放大"到前所未有的尺度}。
\end{bluebox}

\subsection{在日常材料中}

\begin{itemize}
\item \textbf{为什么手机芯片用硅}:带隙 1.12 eV 恰好可调控
\item \textbf{为什么冰箱贴有磁性}:铁氧体的未成对电子(3d)
\item \textbf{为什么 CPU 要散热片}:硅导热有限,需要金属/石墨散热
\item \textbf{为什么 LED 不同颜色}:不同半导体带隙发不同波长的光
\end{itemize}
