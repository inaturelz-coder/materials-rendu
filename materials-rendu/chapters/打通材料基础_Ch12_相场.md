\section{1. 一句话本质}

\begin{bluebox}
\textbf{相场方法 = 用连续的"相场变量"描述组织,通过演化方程让它随时间变化}。
\textbf{它最优雅的地方:界面是"弥散"的(几个网格宽),不用费力追踪尖锐界面在哪}。

\vspace{0.3em}

\textbf{三句话记住这一章}:

\textbf{1. 自由能泛函}:$F = \int[f_{\text{chem}}(c) + \frac{\kappa}{2}|\nabla c|^2]\,\mathrm{d}V$——
\textbf{化学项(来自 CALPHAD)+ 梯度项(界面能)}。

\textbf{2. 两个核心方程}:\textbf{Cahn-Hilliard}(守恒场,如成分)、\textbf{Allen-Cahn}(非守恒场,如相分数)。

\textbf{3. 弥散界面}:界面用平滑过渡描述,\textbf{不用显式追踪界面位置}——数值上极优雅。

\vspace{0.3em}

\textbf{实测核心 Aha}:\textbf{Cahn-Hilliard 方程让均匀的固溶体自发分离成两相(调幅分解,
连接 Ch5 spinodal)——从 $c=0.5$ 演化到贫相 0.015 + 富相 0.993,无需形核};
\textbf{界面宽度 $\propto\sqrt{\kappa/W}$,界面能 $\propto\sqrt{\kappa W}$};
\textbf{相场把"自由能最小化"(静)变成"时间演化"(动)}。
\end{bluebox}

\section{2. 教科书里你看到的}

\subsection{相场变量}

\begin{itemize}
\item \textbf{守恒场}(如成分 $c$):总量守恒(物质不会凭空消失)——用 Cahn-Hilliard
\item \textbf{非守恒场}(如有序参量 $\phi$、相分数):可以局部产生/消失——用 Allen-Cahn
\item \textbf{弥散界面}:相场变量在界面处平滑过渡(不是阶跃)
\end{itemize}

\subsection{两个核心方程}

\begin{longtable}{|l|p{0.62\textwidth}|}
\hline
\rowcolor{primarycolor!10}
\textbf{方程} & \textbf{形式与适用} \\
\hline
Cahn-Hilliard & $\partial c/\partial t = M\nabla^2\mu$,守恒场(成分、调幅分解)\\
Allen-Cahn & $\partial\phi/\partial t = -L\,\delta F/\delta\phi$,非守恒场(界面、有序化)\\
\hline
\end{longtable}

\subsection{自由能泛函}

$$F = \int_V \left[f_{\text{chem}}(c) + \frac{\kappa}{2}|\nabla c|^2\right]\mathrm{d}V$$

\begin{itemize}
\item \textbf{$f_{\text{chem}}$}:化学自由能密度(双势阱,或来自 CALPHAD)
\item \textbf{$\frac{\kappa}{2}|\nabla c|^2$}:梯度能项(惩罚陡峭界面 → 界面能)
\end{itemize}

\subsection{教科书不讲的事}

\begin{itemize}
\item \textbf{为什么"弥散界面"是数值上的革命}(不用追踪界面位置)
\item \textbf{Cahn-Hilliard 如何模拟调幅分解}(连接 Ch5 的 spinodal)
\item \textbf{相场的自由能就是 CALPHAD 的自由能}(静动如何共享地基)
\item \textbf{界面宽度和界面能怎么由参数控制}($\kappa$ 和 $W$)
\end{itemize}

\section{3. 但其实是什么意思}

\subsection{尖锐界面的噩梦,弥散界面的优雅}

\textbf{传统模拟组织演化,要"追踪界面"——界面在哪、怎么移动、合并分裂时怎么处理}。
\textbf{这在数值上是噩梦}:界面是数学上的零厚度面,合并、分裂、形成复杂拓扑时极难处理。

\textbf{相场的革命性想法}:\textbf{不追踪界面,而是用一个连续场 $\phi$ 描述整个系统}。
\textbf{$\phi=1$ 是相 A,$\phi=0$ 是相 B,界面处 $\phi$ 平滑地从 1 过渡到 0(弥散界面)}。
\textbf{界面在哪?就是 $\phi=0.5$ 的等值面——但你不用显式去找它,方程自己处理}。

\textbf{这让相场能轻松模拟最复杂的组织演化}:枝晶生长的复杂分叉、晶粒长大的拓扑变化、
析出相的合并粗化——\textbf{统统不用特殊处理,一套方程搞定}。

\subsection{Cahn-Hilliard:调幅分解的电影}

\textbf{回到 Ch5 的 spinodal(调幅分解)}:\textbf{一个均匀的过饱和固溶体,
在 spinodal 区内会自发分离成两相,无需形核}。\textbf{Ch5 只告诉你"会分离",
Cahn-Hilliard 让你看见"怎么分离"}。

\textbf{方程}:$\dfrac{\partial c}{\partial t} = M\nabla^2\mu$,其中 $\mu = \dfrac{\partial f}{\partial c} - \kappa\nabla^2 c$。
\textbf{从均匀 $c=0.5$ + 微小噪声出发,浓度涨落被放大,自发长出迷宫状的两相组织}——
\textbf{贫相和富相交织}。\textbf{这正是调幅分解的标志性"迷宫"花纹}。

\subsection{Allen-Cahn:界面自己变光滑}

\textbf{Allen-Cahn 描述非守恒场}(如相分数、有序参量)。\textbf{把一个尖锐的阶跃界面交给它,
界面会自动"松弛"成弥散的平衡界面}——\textbf{宽度由 $\kappa$ 和 $W$ 决定}。

\textbf{这就是晶界迁移、有序-无序转变、再结晶的数学描述}。\textbf{驱动力是降低界面能:
弯曲的界面会变直,小晶粒会被大晶粒吞并(晶粒长大)}。

\subsection{相场的自由能 = CALPHAD 的自由能}

\textbf{这是静与动连接的关键}。\textbf{相场自由能泛函里的化学项 $f_{\text{chem}}(c)$,
正是 CALPHAD(Ch7)算出来的那个 $G(x,T)$}。

\begin{bluebox}
\textbf{完整的闭环}:\textbf{CALPHAD(Ch7,静)提供化学自由能 $f_{\text{chem}}$ 和相平衡;
相场(本章,动)用这个自由能 + 梯度项,算出组织如何随时间演化}。
\textbf{CALPHAD 说"该去哪",相场说"怎么去、长什么样"}——\textbf{这是现代 ICME
(集成计算材料工程)的核心,也是"一静一动"在计算上的完美合体}。
\end{bluebox}

\section{4. 真正的数学}

\subsection{自由能泛函与变分}

$$F[c] = \int_V\left[f_{\text{chem}}(c) + \frac{\kappa}{2}|\nabla c|^2\right]\mathrm{d}V$$

\textbf{化学势是泛函导数}:$\mu = \dfrac{\delta F}{\delta c} = \dfrac{\partial f_{\text{chem}}}{\partial c} - \kappa\nabla^2 c$。

\subsection{Cahn-Hilliard 方程(守恒)}

$$\frac{\partial c}{\partial t} = \nabla\cdot(M\nabla\mu) = M\nabla^2\left(\frac{\partial f}{\partial c} - \kappa\nabla^2 c\right)$$

\textbf{注意是 $\nabla^2\mu$——保证总量守恒(成分不会凭空产生)}。\textbf{这是个四阶偏微分方程}。

\subsection{Allen-Cahn 方程(非守恒)}

$$\frac{\partial\phi}{\partial t} = -L\frac{\delta F}{\delta\phi} = -L\left(\frac{\partial f}{\partial\phi} - \kappa\nabla^2\phi\right)$$

\textbf{直接正比于自由能下降方向——非守恒(相分数可以变)}。\textbf{这是二阶方程}。

\subsection{界面宽度与界面能}

\textbf{对双势阱 $f = Wc^2(1-c)^2$,平衡界面的解析结果}:
$$\delta \propto \sqrt{\frac{\kappa}{W}}, \qquad \gamma \propto \sqrt{\kappa W}$$

\textbf{$\kappa$ 大则界面宽、能量高;调节 $\kappa$ 和 $W$ 可以匹配真实材料的界面能}(可由 DFT 算)。

\section{5. 一个让人 "Aha" 的例子}

\subsection{核心 Aha:让均匀固溶体自发分离成两相}

运行配套模块 \texttt{phase\_field.py} 的 Cahn-Hilliard 模拟:

\begin{verbatim}
初始: 均匀 c≈0.5(+小噪声)
演化 8000 步后:
  最低浓度 c_min = 0.015(贫相,趋向0)
  最高浓度 c_max = 0.993(富相,趋向1)
  贫相区域 48%, 富相区域 48%, 界面 3%
\end{verbatim}

\textbf{Aha}:\textbf{从一个几乎均匀的固溶体($c=0.5$ + 微小噪声)出发,
Cahn-Hilliard 方程让它自发分离成贫相($c\to0$)和富相($c\to1$)}——\textbf{无需形核}。
\textbf{这就是 Ch5 讲的"调幅分解"(spinodal),Ch5 只告诉你"会分离",
相场让你"看见分离的全过程"}。\textbf{静(Ch5 说会分)和动(本章演示怎么分)在这里接上了}。

\textbf{现实意义}:\textbf{很多合金的强化相、光学材料的微结构、甚至某些不锈钢的脆化(475°C 脆性),
都是调幅分解的产物}。\textbf{相场能预测它们的尺度和形貌}。

\subsection{弥散界面:相场的优雅}

\begin{verbatim}
Allen-Cahn:初始阶跃界面 → 演化后弥散界面,宽度 ≈ 3 个网格
界面宽度 δ ∝ √(κ/W),界面能 γ ∝ √(κ·W)
κ(梯度能)  界面宽度   界面能
0.5        0.500     1.000
2.0        1.000     2.000
4.0        1.414     2.828
\end{verbatim}

\textbf{Aha}:\textbf{相场不追踪尖锐界面,而是让界面"弥散"成几个网格宽的平滑过渡}。
\textbf{界面宽度和界面能由参数 $\kappa$、$W$ 控制——而界面能 $\gamma$ 可以由 DFT 算(连接 Ch3/Ch9)}。
\textbf{这种"弥散界面"是相场数值优雅的根源}:\textbf{枝晶分叉、晶粒吞并、析出粗化等复杂拓扑变化,
不需要任何特殊处理,方程自动搞定}。

\subsection{怎么看见它:相场模拟 vs 实验组织对照}

\begin{bluebox}
\textbf{怎么看见它}。\textbf{相场是"计算"——它的"看见"是与实验组织(金相、SEM、3D 重构)对照验证}。
\end{bluebox}

\textbf{相场模拟的验证手段}:
\begin{itemize}
\item \textbf{金相/SEM(Ch4)}:把模拟的组织形貌(枝晶、晶粒、析出)与实验照片对比——\textbf{形貌、尺度是否一致}
\item \textbf{原位实验}:同步辐射 X 射线断层(in-situ tomography)实时拍组织演化——\textbf{直接对照相场"电影"}
\item \textbf{3D 组织重构}:FIB 连续切片 + EBSD 建真实 3D 组织,作相场初始条件或验证
\item \textbf{统计量对比}:晶粒尺寸分布、相体积分数(Ch4 的定量金相)、界面面积——\textbf{统计意义上验证}
\end{itemize}

\textbf{相场的输入从哪来(连接全书)}:\textbf{化学自由能 ← CALPHAD(Ch7);界面能 ← DFT(Ch3/Ch9);
扩散系数 ← 实验/DICTRA(Ch8);弹性常数 ← DFT}。\textbf{能算什么}:枝晶凝固、晶粒长大、
析出与粗化(Ostwald ripening)、调幅分解、马氏体相变、烧结、裂纹扩展。
\textbf{局限}:\textbf{界面要几个网格宽,限制了能模拟的尺度};\textbf{参数需要从别处获取(自由能、界面能、迁移率)};
\textbf{大尺度三维模拟极耗算力(需 GPU/超算)};\textbf{是介观模型,不含原子细节}。

\textbf{你的研究方向}:\textbf{用相场验证 CALPHAD 反演的相图——
让"算出来的相图"经得起"组织演化模拟"的检验}。\textbf{这正是静(相图)与动(相场)
互相印证的前沿}。
